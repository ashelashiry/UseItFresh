Future<void> main(List<String> args) async {
  final options = _parseCliOptions(args);
  try {
    await flutterFlowAI(
      buildStarterEditFlow,
      apiKey: options.apiKey,
      baseUrl: options.baseUrl,
      projectName: options.projectName,
      projectId: options.projectId,
      findOrCreate: options.findOrCreate,
      allowNewProject: options.allowNewProject,
      dryRun: options.dryRun,
      commitMessage: options.commitMessage,
    );
  } catch (error) {
    stderr.writeln('Error: ${formatFlutterFlowAIError(error)}');
    exit(1);
  }
}

final class _CliOptions {
  const _CliOptions({
    this.apiKey,
    this.baseUrl,
    this.projectName,
    this.projectId,
    this.findOrCreate = false,
    this.allowNewProject = false,
    this.dryRun = false,
    this.commitMessage,
  });

  final String? apiKey;
  final String? baseUrl;
  final String? projectName;
  final String? projectId;
  final bool findOrCreate;
  final bool allowNewProject;
  final bool dryRun;
  final String? commitMessage;
}

_CliOptions _parseCliOptions(List<String> args) {
  String? apiKey;
  String? baseUrl;
  String? projectName;
  String? projectId;
  String? commitMessage;
  var findOrCreate = false;
  var allowNewProject = false;
  var dryRun = false;

  for (var i = 0; i < args.length; i++) {
    final arg = args[i];
    switch (arg) {
      case '--help':
      case '-h':
        _printUsage();
        exit(0);
      case '--api-key':
        apiKey = _requireValue(args, ++i, '--api-key');
      case '--base-url':
        baseUrl = _requireValue(args, ++i, '--base-url');
      case '--project-name':
        projectName = _requireValue(args, ++i, '--project-name');
      case '--project-id':
        projectId = _requireValue(args, ++i, '--project-id');
      case '--commit-message':
        commitMessage = _requireValue(args, ++i, '--commit-message');
      case '--find-or-create':
        findOrCreate = true;
      case '--allow-new-project':
        allowNewProject = true;
      case '--dry-run':
        dryRun = true;
      default:
        stderr.writeln('Unknown option: $arg');
        _printUsage();
        exit(64);
    }
  }

  return _CliOptions(
    apiKey: apiKey,
    baseUrl: baseUrl,
    projectName: projectName,
    projectId: projectId,
    findOrCreate: findOrCreate,
    allowNewProject: allowNewProject,
    dryRun: dryRun,
    commitMessage: commitMessage,
  );
}

String _requireValue(List<String> args, int index, String flag) {
  if (index >= args.length) {
    stderr.writeln('Missing value for $flag.');
    _printUsage();
    exit(64);
  }
  return args[index];
}

void _printUsage() {
  stdout.writeln('''
Run the starter FlutterFlow AI edit flow.

Usage:
  dart run dsl/edit.dart [options]

Options:
  --api-key <key>           FlutterFlow API key. Defaults to FF_API_KEY.
  --base-url <url>          Override the FlutterFlow API base URL.
  --project-name <name>     Create a new project with this name.
  --project-id <id>         Push into an existing project by ID.
  --find-or-create          Retry by reusing a same-name project before creating.
  --allow-new-project       Bypass the workspace binding guard and create a different project.
  --commit-message <text>   Commit message for the push.
  --dry-run                 Compile and validate without pushing.
  --help, -h                Show this help.
''');
}

// ---------------------------------------------------------------------------
// The entrance:  fridge appears -> shimmers -> tap -> door opens -> login fades in
//
// Both clips are the +faststart re-encodes. The originals had their moov atom
// at the END of the file, so a browser had to download the whole thing before
// showing a frame. The 96KB glint scraped through; the 993KB opening did not,
// which is why tapping did nothing.
// ---------------------------------------------------------------------------

/// Closed fridge with the handle glint, looping. The resting state.
const _kGlintLoop =
    'https://storage.googleapis.com/flutterflow-io-6f20.appspot.com'
    '/projects/fridge-wise-gvpy0s/assets/2dj64a1ej31u/web-fridge-closed-glint-loop.mp4';

/// The door opening. 2.13s.
const _kOpeningClip =
    'https://storage.googleapis.com/flutterflow-io-6f20.appspot.com'
    '/projects/fridge-wise-gvpy0s/assets/daf0i051xyda/web-fridge-opening-v7.mp4';

/// The glint clip's first frame, as a still. Sits UNDER the video so the very
/// first paint is already a closed fridge — an image draws immediately, a video
/// does not, and without this you saw straight through to the open fridge
/// behind for a moment on load.
const _kClosedPoster =
    'https://storage.googleapis.com/flutterflow-io-6f20.appspot.com'
    '/projects/fridge-wise-gvpy0s/assets/wte62cfd8wp4/fridge-closed-v7.jpg';

/// Clip length plus slack, so the form never lands over a door still moving.
const _kClipMs = 2300;

void buildStarterEditFlow(App app) {
  // ------------------------------------------------------------------
  // 1. Give the three flags real defaults.
  //
  // This is the bug behind "the first glimpse is an open fridge, then
  // nothing happens". FlutterFlow generates `if (_model.formShown ?? true)`
  // for a visibility binding, so a page-state bool with NO default is null
  // on the first frame and every gated layer renders at once — open fridge,
  // opening clip and sign-in form stacked on top of each other — until the
  // page-load actions run a frame or two later.
  //
  // With defaults the very first paint is already the correct state and no
  // page-load action is needed to get there.
  // ------------------------------------------------------------------
  app.editPageState(ff.Pages.signInPage, (state) {
    state.ensureField('introPending', bool_.withDefault(true));
    state.ensureField('introPlaying', bool_.withDefault(false));
    state.ensureField('formShown', bool_.withDefault(false));
  });

  // The intro now runs whenever the sign-in page is shown, not once per
  // install. A signed-in user never reaches this page — they land on Home —
  // so "every time" and "first time" only differ for someone who signed out,
  // and showing them the door opening again is the better of the two. It also
  // makes the thing testable, which hasOpenedFridge did not.
  app.editPageOnLoad(ff.Pages.signInPage, [
    SetState(ff.Pages.signInPage.state.introPending, true),
    SetState(ff.Pages.signInPage.state.introPlaying, false),
    SetState(ff.Pages.signInPage.state.formShown, false),
  ]);

  app.editPage(ff.Pages.signInPage, (page) {
    // ----------------------------------------------------------------
    // 2. Both clips point at the +faststart re-encodes.
    //
    // The originals carry their moov atom at the END of the file, so a
    // browser has to fetch the whole thing before it can show frame one.
    // The 96KB glint scraped through; the 993KB opening did not, which is
    // why tapping appeared to do nothing.
    //
    // Written through mutateNode because EditWidgetPatch has no path setter.
    // ----------------------------------------------------------------
    page.mutateNode(
      ff.Pages.signInPage.widgets.byKey('VideoPlayer_ajavvhaa').single,
      (node) {
        node.props.videoPlayer.pathValue = FFStringValue(
          inputValue: _kOpeningClip,
        );
      },
    );

    // 3. The open-fridge backdrop stops rendering while the door is still
    //    closed. It sits at the bottom of the Stack and was showing through
    //    for the moment before the closed layer painted.
    page.bindVisible(
      ff.Pages.signInPage.widgets.byKey('Image_sdgiq17v').single,
      Not(State(ff.Pages.signInPage.state.introPending)),
    );

    // 4. Resting state: the closed still with the looping glint over it.
    //    An image paints on the first frame, a video does not, so the still
    //    underneath is what removes the last of the flash.
    page.ensureReplaced(
      ff.Pages.signInPage.widgets.byKey('VideoPlayer_b93v1rf2').single,
      Stack(
        name: 'ClosedFridgeLayers',
        alignment: Alignment.center,
        children: [
          Image(
            _kClosedPoster,
            name: 'ClosedFridgePoster',
            fit: ImageFit.cover,
            width: double.infinity,
            height: double.infinity,
          ),
          VideoPlayer(
            _kGlintLoop,
            name: 'ClosedFridgeGlint',
            autoPlay: true,
            looping: true,
            showControls: false,
            width: double.infinity,
            height: double.infinity,
          ),
        ],
      ),
    );

    // 5. Tap -> play the opening -> fade the form in.
    page.ensureActions(
      ff.Pages.signInPage.widgets.byKey('Container_o5pu6w5t').single,
      triggerType: FFActionTriggerType.ON_TAP,
      actions: [
        SetState(ff.Pages.signInPage.state.introPending, false),
        SetState(ff.Pages.signInPage.state.introPlaying, true),
        const Wait(_kClipMs),
        SetState(ff.Pages.signInPage.state.introPlaying, false),
        SetState(ff.Pages.signInPage.state.formShown, true),
        // SignInFormPanel already carries an onActionTrigger fade+slide;
        // this is what fires it. The key it points at is filled in below.
        const StartAnimation('SignInFormPanel'),
      ],
    );
  });

  // ------------------------------------------------------------------
  // 6. Point the StartAnimation at the panel.
  //
  // StartAnimation resolves its target by walking the page tree in a
  // DEFERRED fixup that only runs inside the greenfield compile() pass.
  // The edit path (page.ensureActions) compiles the chain and attaches it
  // without ever draining that queue, so the action ships with an empty
  // animationNodeKeyRef and the server rejects the push with
  // "has an Animation action but the animation widget is not set".
  //
  // Raw mutations run in registration order after the editPage above, so
  // by the time this runs the chain is on the node and the key can be
  // filled in directly.
  // ------------------------------------------------------------------
  app.raw((project) {
    final page = findPage(project, name: 'SignInPage');
    if (page == null) return;
    final panel = findByKey(page.node, 'Container_yyjai4fl');
    final tapTarget = findByKey(page.node, 'Container_o5pu6w5t');
    if (panel == null || tapTarget == null) return;
    for (final trigger in tapTarget.triggerActions) {
      _fillAnimationTarget(trigger.rootAction, panel.key);
    }
  });
}

/// Walks an action chain and gives every widget-animation action with an
/// unset target the supplied node key.
void _fillAnimationTarget(FFActionNode node, String targetKey) {
  if (node.hasAction() && node.action.hasAnimation()) {
    final anim = node.action.animation;
    if (anim.hasWidgetAnimation() &&
        anim.widgetAnimation.animationNodeKeyRef.key.isEmpty) {
      anim.widgetAnimation.animationNodeKeyRef = FFNodeKeyReference(
        key: targetKey,
      );
    }
  }
  if (node.hasFollowUpAction()) {
    _fillAnimationTarget(node.followUpAction, targetKey);
  }
}
