void buildStarterEditFlow(App app) {
  // ------------------------------------------------------------------
  // Blocker #3 — the reset-password flow dead-ended: Supabase sends a
  // recovery link, but the project had no Update Password screen for the
  // client to land on. This adds it.
  //
  // The action chain is deliberately JUST UpdatePassword. FlutterFlow's
  // generated SupabaseAuthManager.updatePassword already shows its own
  // snackbar on both paths ('Password updated successfully' / 'Error: ...')
  // and does NOT halt the chain on failure — so any trailing Snackbar or
  // NavigateBack here would fire on failure too and lie to the user.
  // The confirm-password check is enforced by an `If` in the chain — see the
  // note on the button.
  // ------------------------------------------------------------------
  final updatePasswordPage = app.ensurePage(
    'UpdatePasswordPage',
    route: '/update-password',
    description:
        'Sets a new password — the landing screen for Supabase recovery '
        'links and the change-password entry point in Profile.',
    state: {
      'password': string.withDefault(''),
      'confirmPassword': string.withDefault(''),
    },
    body: Scaffold(
      appBar: AppBar(title: 'Update password'),
      body: Container(
        padding: 24,
        child: Column(
          spacing: 16,
          crossAxis: CrossAxis.start,
          scrollable: true,
          children: [
            Text(
              'Choose a new password. Use at least 8 characters.',
              style: Styles.bodyMedium,
              color: Colors.secondaryText,
              maxLines: 2,
            ),
            TextField(
              name: 'UpdatePasswordField',
              label: 'New password',
              obscureText: true,
              onChanged: SetState(ff.Pages.updatePasswordPage.state.password, const TextValue()),
            ),
            TextField(
              name: 'ConfirmUpdatePasswordField',
              label: 'Confirm new password',
              obscureText: true,
              onChanged: SetState(ff.Pages.updatePasswordPage.state.confirmPassword, const TextValue()),
            ),
            Button(
              'Update password',
              name: 'UpdatePasswordButton',
              width: double.infinity,
              borderRadius: 12,
              onTap: [_updatePasswordChain()],
            ),
          ],
        ),
      ),
    ),
  );

  // ensurePage(...) no-ops once the page exists, so the chain above is only
  // written on the very first run. Re-apply it through the edit surface so
  // subsequent runs converge on the same wiring.
  app.editPage(ff.Pages.updatePasswordPage, (page) {
    page.ensureActions(
      ff.Pages.updatePasswordPage.widgets.byKey('Button_wmi81uaz').single,
      triggerType: FFActionTriggerType.ON_TAP,
      actions: [_updatePasswordChain()],
    );
  });

  // ------------------------------------------------------------------
  // Blocker #2 — SendResetLinkButton claimed success unconditionally.
  //
  // The shipped chain was [ResetPassword, Snackbar('Reset email sent'),
  // NavigateBack]. SupabaseAuthManager.resetPassword catches AuthException,
  // shows 'Error: <message>' and returns — it does not stop the chain. So on
  // failure the user saw the real error flash, then a success message, then
  // got popped off the page. Reducing the chain to ResetPassword alone leaves
  // the manager's own truthful snackbar as the single source of truth (it
  // already emits 'Password reset email sent' on success) and keeps the user
  // on the page so they can retry. Codegen still emits the empty-email guard.
  // ------------------------------------------------------------------
  app.editPage(ff.Pages.resetPasswordPage, (page) {
    page.ensureActions(
      ff.Pages.resetPasswordPage.widgets.byKey('Button_d4zdk87a').single,
      triggerType: FFActionTriggerType.ON_TAP,
      actions: [ResetPassword(State('email'))],
    );
  });

  // ------------------------------------------------------------------
  // Reachability for the new page: a signed-in user can change their
  // password from Profile, so the screen is not reachable only via a
  // recovery deep link.
  // ------------------------------------------------------------------
  app.editPage(ff.Pages.profilePage, (page) {
    page.ensureInsertedAfter(
      ff.Pages.profilePage.widgets.byKey('Button_s1lf9l7y').single,
      Column(
        name: 'SecuritySection',
        spacing: 12,
        crossAxis: CrossAxis.start,
        children: [
          ListTile(
            title: 'Security',
            subtitle: 'Change your password',
            leadingIcon: 'lock',
            trailingIcon: 'chevron_right',
          ),
          Button(
            'Change password',
            name: 'ChangePasswordButton',
            variant: ButtonVariant.outlined,
            width: double.infinity,
            borderRadius: 12,
            onTap: Navigate(updatePasswordPage),
          ),
        ],
      ),
    );
  });
}

/// The UpdatePasswordButton ON_TAP chain.
///
/// The confirm-password check has to live here rather than on the action:
/// FlutterFlow's codegen for UPDATE_PASSWORD ignores confirm_password_variable
/// (unlike SIGNUP, which does emit a mismatch guard from it), so passing it
/// alone leaves the confirm field decorative. `If` emits the modern
/// true_actions[0] shape.
DslAction _updatePasswordChain() => If(
  Equals(
    State(ff.Pages.updatePasswordPage.state.password),
    State(ff.Pages.updatePasswordPage.state.confirmPassword),
  ),
  then: [
    UpdatePassword(
      State(ff.Pages.updatePasswordPage.state.password),
      confirmPassword: State(
        ff.Pages.updatePasswordPage.state.confirmPassword,
      ),
    ),
  ],
  orElse: [Snackbar('Passwords do not match.')],
);
