Future<void> main(List<String> args) async {
  final options = _parseCliOptions(args);
  try {
    await flutterFlowAI(
      buildStarterEditFlow,
      apiKey: options.apiKey,
      baseUrl: options.baseUrl,
      projectName: options.projectName,
      projectId: options.projectId,
      findOrCreate: options.findOrCreate,
      allowNewProject: options.allowNewProject,
      dryRun: options.dryRun,
      commitMessage: options.commitMessage,
    );
  } catch (error) {
    stderr.writeln('Error: ${formatFlutterFlowAIError(error)}');
    exit(1);
  }
}

final class _CliOptions {
  const _CliOptions({
    this.apiKey,
    this.baseUrl,
    this.projectName,
    this.projectId,
    this.findOrCreate = false,
    this.allowNewProject = false,
    this.dryRun = false,
    this.commitMessage,
  });

  final String? apiKey;
  final String? baseUrl;
  final String? projectName;
  final String? projectId;
  final bool findOrCreate;
  final bool allowNewProject;
  final bool dryRun;
  final String? commitMessage;
}

_CliOptions _parseCliOptions(List<String> args) {
  String? apiKey;
  String? baseUrl;
  String? projectName;
  String? projectId;
  String? commitMessage;
  var findOrCreate = false;
  var allowNewProject = false;
  var dryRun = false;

  for (var i = 0; i < args.length; i++) {
    final arg = args[i];
    switch (arg) {
      case '--help':
      case '-h':
        _printUsage();
        exit(0);
      case '--api-key':
        apiKey = _requireValue(args, ++i, '--api-key');
      case '--base-url':
        baseUrl = _requireValue(args, ++i, '--base-url');
      case '--project-name':
        projectName = _requireValue(args, ++i, '--project-name');
      case '--project-id':
        projectId = _requireValue(args, ++i, '--project-id');
      case '--commit-message':
        commitMessage = _requireValue(args, ++i, '--commit-message');
      case '--find-or-create':
        findOrCreate = true;
      case '--allow-new-project':
        allowNewProject = true;
      case '--dry-run':
        dryRun = true;
      default:
        stderr.writeln('Unknown option: $arg');
        _printUsage();
        exit(64);
    }
  }

  return _CliOptions(
    apiKey: apiKey,
    baseUrl: baseUrl,
    projectName: projectName,
    projectId: projectId,
    findOrCreate: findOrCreate,
    allowNewProject: allowNewProject,
    dryRun: dryRun,
    commitMessage: commitMessage,
  );
}

String _requireValue(List<String> args, int index, String flag) {
  if (index >= args.length) {
    stderr.writeln('Missing value for $flag.');
    _printUsage();
    exit(64);
  }
  return args[index];
}

void _printUsage() {
  stdout.writeln('''
Run the starter FlutterFlow AI edit flow.

Usage:
  dart run dsl/edit.dart [options]

Options:
  --api-key <key>           FlutterFlow API key. Defaults to FF_API_KEY.
  --base-url <url>          Override the FlutterFlow API base URL.
  --project-name <name>     Create a new project with this name.
  --project-id <id>         Push into an existing project by ID.
  --find-or-create          Retry by reusing a same-name project before creating.
  --allow-new-project       Bypass the workspace binding guard and create a different project.
  --commit-message <text>   Commit message for the push.
  --dry-run                 Compile and validate without pushing.
  --help, -h                Show this help.
''');
}

// ---------------------------------------------------------------------------
// Home: greet by name, and show what actually needs eating.
//
// Two things were placeholders. The greeting printed the account's email
// address, which is not what anyone calls themselves — profiles.display_name
// has existed since the first migration and was never read. And "Use first",
// which is the entire point of the app, was a permanent empty-state card even
// once there was food in the fridge.
//
// The list reads food_items_status, the VIEW from migration 4, not the
// food_items table. The view recomputes the §10 status against today's date
// inside Postgres and carries the label, the plain-English detail and the
// reasoning with it, so the order is always current and the row never
// re-derives what the database already worked out.
//
// Values reach the widgets as BINDINGS, not by being copied into page state
// first. Reading a field off a query result and storing it with SetState is
// rejected — "update value that is not properly set" — because a row field is
// only resolvable where it is displayed. So page state holds whole rows, and
// the field access happens at the widget.
// ---------------------------------------------------------------------------

/// Top three only. Home is a glance, not the inventory — the full list, in the
/// same order, is one tap away on the Inventory tab.
const _kUseFirstLimit = 3;

/// Urgency first, then how many days are left. Both are computed columns on
/// the view, so Postgres sorts against today's date rather than the app
/// sorting a snapshot that was true when the screen last loaded.
const _kUrgencyOrder = <PostgresOrderBy>[
  PostgresOrderBy('urgency_rank'),
  PostgresOrderBy('days_left'),
];

void buildStarterEditFlow(App app) {
  // Someone who signed up two minutes ago has no display name yet, and
  // "Welcome back," with nothing after it is worse than the email address.
  //
  // It takes the whole row LIST rather than the name, because a single-row
  // result cannot be stored in page state — "update value that is not
  // properly set" — while a list can. So the query keeps its list shape all
  // the way to here and Dart picks the row.
  final greetingName = app.customFunction(
    'greetingName',
    args: {'rows': listOf(ff.Tables.profiles), 'email': string},
    returns: string,
    code: r'''
if (rows != null && rows.isNotEmpty) {
  final name = rows.first.displayName?.trim() ?? '';
  if (name.isNotEmpty) return name;
}
final address = email ?? '';
// Not a real name, but better than an empty greeting: the part before the @
// is what most people would recognise as theirs.
final at = address.indexOf('@');
return at > 0 ? address.substring(0, at) : address;
''',
    description: 'Display name if there is one, otherwise something from the '
        'email address.',
  );

  // There is no "list is empty" expression in the DSL, and a row field cannot
  // stand in for one — an absent row makes the field null, not empty, so the
  // comparison silently reads false and the empty card never appears.
  final hasRows = app.customFunction(
    'hasRows',
    args: {'rows': listOf(ff.Tables.foodItemsStatus)},
    returns: bool_,
    code: 'return rows != null && rows.isNotEmpty;',
    description: 'True when a food_items_status query returned anything.',
  );

  // ---------------------------- Home ----------------------------
  app.editPageState(ff.Pages.homePage, (state) {
    state.ensureField('me', listOf(ff.Tables.profiles));
    state.ensureField('useFirst', listOf(ff.Tables.foodItemsStatus));
  });

  app.editPageOnLoad(ff.Pages.homePage, [
    // RLS already restricts profiles to the signed-in user, but the filter is
    // explicit anyway — a query that only works because of a policy is one
    // that breaks silently the day the policy changes.
    PostgresQuery(
      ff.Tables.profiles,
      outputAs: 'meRow',
      query: PostgresQuerySpec(
        limit: 1,
        filters: [
          PostgresFilter(
            'id',
            relation: PostgresFilterRelation.equalTo,
            value: const AuthUser(AuthUserField.userId),
          ),
        ],
      ),
    ),
    SetState('me', ActionOutput('meRow')),
    PostgresQuery(
      ff.Tables.foodItemsStatus,
      outputAs: 'urgentRows',
      query: PostgresQuerySpec(
        orderBys: _kUrgencyOrder,
        limit: _kUseFirstLimit,
      ),
    ),
    SetState('useFirst', ActionOutput('urgentRows')),
  ]);

  app.editPage(ff.Pages.homePage, (page) {
    page.bindText(
      ff.Pages.homePage.widgets.byKey('Text_00bqjarl').single,
      CustomFunction(
        greetingName,
        args: {
          'rows': State('me'),
          'email': const AuthUser(AuthUserField.email),
        },
      ),
    );

    page.ensureReplaced(
      ff.Pages.homePage.widgets.byKey('Container_p36qqddg').single,
      Column(
        name: 'UseFirstSection',
        crossAxis: CrossAxis.stretch,
        spacing: 10,
        children: [
          ListView(
            name: 'UseFirstList',
            source: State('useFirst'),
            spacing: 10,
            // Home scrolls as one page; a list with its own scroll inside a
            // scrolling column swallows the gesture.
            shrinkWrap: true,
            scrollPhysics: ScrollPhysics.never,
            itemBuilder: (item) => ff.Components.foodCard(
              name: 'UseFirstCard',
              itemName: item['name'],
              // Where to go and look for it.
              meta: item['location_name'],
              // Plain English, computed by the view — "use today", "3 days
              // left". Never a raw date, and never colour on its own (§10).
              dateLine: item['status_detail'],
              statusLabel: item['status_label'],
            ),
          ),
          ff.Components.emptyStateCard(
            name: 'UseFirstEmpty',
            title: 'Nothing to use up yet',
            message: 'Add what is in your fridge and whatever is closest to '
                'its date will show up here.',
            visible: Not(CustomFunction(hasRows, args: {'rows': State('useFirst')})),
          ),
        ],
      ),
    );
  });

  // -------------------------- Inventory --------------------------
  // The inventory's empty hint had the same bug in reverse: it sat under the
  // list permanently, telling you there was nothing here while listing things.
  app.editPage(ff.Pages.inventoryPage, (page) {
    page.bindVisible(
      ff.Pages.inventoryPage.widgets.byKey('Text_v3h0m6fk').single,
      Not(CustomFunction(hasRows, args: {'rows': State('items')})),
    );
  });
}
