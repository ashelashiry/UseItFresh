"""What each screen does when the signal drops.

Kitchens and shops are where signal goes. This walk opens the app online, in a
household of its own with a few foods in it, then cuts the browser's network
(CDP Network.emulateNetworkConditions) and visits the screens that load from
the database. Each screenshot is what a person would see; each error the page
logs is recorded against the screen it came from. Then it goes back online to
see whether the app recovers without a restart.

Same safety as every walk since 11 Sep: its own household, found by diffing
ids, seeded through the API and deleted at the end by id AND temp name.
Nothing is written while offline.
"""

problems = []
STEP = ["setup"]


def step(label):
    STEP[0] = label
    print(label)


def send(ws, m, p=None, _i=[0]):
    _i[0] += 1
    ws.send(json.dumps({"id": _i[0], "method": m, "params": p or {}}))
    return _i[0]


def pump(ws, seconds):
    end = time.time() + seconds
    ws.settimeout(0.4)
    while time.time() < end:
        try:
            msg = json.loads(ws.recv())
        except Exception:
            continue
        m, p = msg.get("method", ""), msg.get("params", {})
        if m == "Runtime.exceptionThrown":
            d = p["exceptionDetails"]
            problems.append("[%s] EXCEPTION: %s %s" % (
                STEP[0], d.get("text") or "",
                ((d.get("exception") or {}).get("description") or "")[:200]))
        elif m == "Runtime.consoleAPICalled" and p.get("type") in ("error", "warning"):
            txt = " ".join(str(a.get("value", a.get("description", "")))
                           for a in p.get("args", []))
            if "app-icon-512" not in txt:
                problems.append("[%s] CONSOLE[%s]: %s" % (STEP[0], p.get("type"), txt[:300]))
        elif m == "Network.loadingFailed":
            problems.append("[%s] NETWORK FAILED: %s" % (STEP[0], p.get("errorText")))


def shot(ws, name=None, tries=2):
    for attempt in range(tries):
        rid = send(ws, "Page.captureScreenshot", {"format": "png"})
        ws.settimeout(30)
        try:
            while True:
                msg = json.loads(ws.recv())
                if msg.get("id") == rid:
                    im = Image.open(io.BytesIO(base64.b64decode(
                        msg["result"]["data"]))).convert("RGB")
                    if name:
                        im.save(os.path.join(OUT, name))
                        print("   saved", name)
                    return im
        except websocket.WebSocketTimeoutException:
            print("   (screenshot timed out, attempt %d)" % (attempt + 1))
    raise RuntimeError("no screenshot after %d tries" % tries)


def click(ws, x, y, settle=1.2):
    for k in ("mousePressed", "mouseReleased"):
        send(ws, "Input.dispatchMouseEvent",
             {"type": k, "x": int(x), "y": int(y), "button": "left",
              "clickCount": 1, "buttons": 1 if k == "mousePressed" else 0})
    pump(ws, settle)


def type_text(ws, text):
    for ch in text:
        send(ws, "Input.dispatchKeyEvent", {"type": "keyDown", "key": ch})
        send(ws, "Input.dispatchKeyEvent", {"type": "char", "text": ch, "key": ch})
        send(ws, "Input.dispatchKeyEvent", {"type": "keyUp", "key": ch})
        pump(ws, 0.03)


def green_rows(im, lo, colour=(37, 102, 81), tol=22, xs=range(120, 320, 5), need=28):
    px = im.load()
    rows = [y for y in range(lo, im.size[1])
            if sum(1 for x in xs
                   if abs(px[x, y][0] - colour[0]) < tol
                   and abs(px[x, y][1] - colour[1]) < tol
                   and abs(px[x, y][2] - colour[2]) < tol) > need]
    if not rows:
        return None
    runs, run = [], [rows[0]]
    for y in rows[1:]:
        if y - run[-1] > 2:
            runs.append(run)
            run = []
        run.append(y)
    runs.append(run)
    best = max(runs, key=len)
    return (best[0] + best[-1]) // 2


def offline(ws, on):
    send(ws, "Network.emulateNetworkConditions",
         {"offline": on, "latency": 0, "downloadThroughput": -1,
          "uploadThroughput": -1})
    pump(ws, 1.0)
